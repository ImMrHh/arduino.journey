Code Challenge

Fill in the blanks to complete the LED sequence:

const int NUM_LEDS = ___;
int ledPins[] = {3, ___, 5};

void setup() {
  for (int i = 0; i < ___; i++) {
    pinMode(ledPins[i], OUTPUT);
  }
}

void loop() {
  for (int i = 0; i < NUM_LEDS; ___) {
    digitalWrite(ledPins[i], HIGH);
    delay(300);
    digitalWrite(ledPins[___], LOW);
  }
}
